Scenario 4: Security Threat (Brute-Force Script)

# 1. First, get your backend service IP or port-forward
kubectl port-forward svc/backend-service 5000:5000 -n demo

# 2. Run the brute force in a separate PowerShell tab
$baseUrl = "http://localhost:5000"
$protectedEndpoints = @("/api/admin/dashboard", "/api/orders")

Write-Host "Initiating Admin API Brute-Force Simulation..." -ForegroundColor Red
1..50 | ForEach-Object {
    $endpoint = $protectedEndpoints | Get-Random
    $targetUrl = "$baseUrl$endpoint"
    $fakeKey = "admin-key-guess-$_"
    
    try {
        Invoke-RestMethod -Uri $targetUrl -Headers @{"X-Admin-Key"="$fakeKey"} -ErrorAction SilentlyContinue | Out-Null
    } catch {
        Write-Host "Unauthorized Access Attempt: $targetUrl with key '$fakeKey' - Blocked" -ForegroundColor Yellow
    }
    Start-Sleep -Milliseconds 150 
}
